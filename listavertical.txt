package com.example.myapplication;

import android.os.Bundle;
import android.view.View;
import android.widget.AdapterView;
import android.widget.ListView;
import android.widget.Toast;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;

// Mantenemos ListView y quitamos el RecyclerView que causaba conflicto
public class MainActivity extends AppCompatActivity implements AdapterView.OnItemClickListener {

    String[] deportescomenta = {"uno", "dos", "tres", "cuatro", "cinco", "seis", "siete", "ocho", "nueve", "diez"};
    String[] deportesLista = {"depo1", "depo2", "depo3", "depo4", "depo5", "depo6", "depo7", "depo8", "depo9", "depo10"};
    int[] deporteImagenes = {R.drawable.img1, R.drawable.img2, R.drawable.img3, R.drawable.img4, R.drawable.img5, R.drawable.img6, R.drawable.img7, R.drawable.img8, R.drawable.img9, R.drawable.img10};

    ListView listView; // Usamos ListView para que coincida con tu Adaptador de BaseAdapter

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);

        listView = findViewById(R.id.vistita);
        listView.setOnItemClickListener(this);

        // CORREGIDO: Usamos los nombres exactos de tus variables declaradas arriba
        Adaptador adaptador = new Adaptador(this, deportesLista, deportescomenta, deporteImagenes);
        listView.setAdapter(adaptador);
    }

    @Override
    public void onItemClick(AdapterView<?> parent, View view, int position, long id) {
        Toast.makeText(this, String.valueOf(position), Toast.LENGTH_SHORT).show();
    }
}
